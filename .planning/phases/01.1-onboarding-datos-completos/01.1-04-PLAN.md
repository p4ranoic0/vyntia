Two-phase upload state pattern (apply to ALL three dialog types):
```typescript
// Dialog local state additions:
const [pendingDoc, setPendingDoc] = useState<{ file: File; tipo: string; label: string } | null>(null)
const [isDocPreviewOpen, setIsDocPreviewOpen] = useState(false)
const [stagedFile, setStagedFile] = useState<File | null>(null)  // file waiting for preview confirm

// File selected/dropped handler:
const handleFileSelected = (file: File) => {
  setStagedFile(file)
  setIsDocPreviewOpen(true)  // open preview
}

// Preview confirm handler:
const handlePreviewConfirm = () => {
  if (stagedFile) setPendingDoc({ file: stagedFile, tipo: 'tipo_documento_here', label: 'Label here' })
  setStagedFile(null)
  setIsDocPreviewOpen(false)
}

// Preview cancel handler:
const handlePreviewCancel = () => {
  setStagedFile(null)
  setIsDocPreviewOpen(false)
}

// Create mutation onSuccess:
onSuccess: async () => {
  if (pendingDoc) {
    try {
      await subirDocumento(empleadoId, pendingDoc.tipo, 'familiar', pendingDoc.label, pendingDoc.file)
      toast.success('Registro y documento guardados')
    } catch {
      toast.warning('Registro guardado. Error al subir documento — puede subirlo luego.')
    }
    setPendingDoc(null)
  } else {
    toast.success('Registro guardado exitosamente')
  }
  // invalidate queries, reset form, close dialog
}

// Dialog close/cancel — reset staged state:
const handleDialogClose = () => {
  setPendingDoc(null)
  setStagedFile(null)
  setIsDocPreviewOpen(false)
  // ... close dialog
}
```
